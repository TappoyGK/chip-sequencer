cmake_minimum_required(VERSION 3.16)
project(chip-sequencer VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
if(NOT TARGET chip-audio::chip-audio)
    find_package(chip-audio CONFIG REQUIRED)
endif()
add_library(chip-sequencer STATIC src/chip-sequencer.c src/compiler.c)
add_library(chip-sequencer::chip-sequencer ALIAS chip-sequencer)
target_compile_features(chip-sequencer PUBLIC c_std_11)
target_include_directories(chip-sequencer PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(chip-sequencer PUBLIC chip-audio::chip-audio)
install(TARGETS chip-sequencer EXPORT chip-sequencer-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT chip-sequencer-targets NAMESPACE chip-sequencer:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chip-sequencer)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/chip-sequencer)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/chip-sequencer-config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/chip-sequencer-config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chip-sequencer)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/chip-sequencer-config-version.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/chip-sequencer-config.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/chip-sequencer-config-version.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chip-sequencer)
option(CHIP_SEQUENCER_BUILD_TESTS "Build sequencer tests" OFF)
if(CHIP_SEQUENCER_BUILD_TESTS)
    enable_testing()
    enable_language(CXX)
    add_executable(cxx-link-test tests/cxx-link-test.cpp)
    target_compile_features(cxx-link-test PRIVATE cxx_std_17)
    target_link_libraries(cxx-link-test PRIVATE chip-sequencer)
    add_test(NAME cxx-link COMMAND cxx-link-test)
    foreach(name sequence compile)
        add_executable(${name}-test tests/${name}-test.c)
        target_link_libraries(${name}-test PRIVATE chip-sequencer)
        if(NOT MSVC)
            target_compile_options(${name}-test PRIVATE -UNDEBUG)
        endif()
    endforeach()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(diagnostic-probe tests/diagnostic-probe.c)
    target_link_libraries(diagnostic-probe PRIVATE chip-sequencer)
    if(MSVC)
        target_compile_options(diagnostic-probe PRIVATE /UNDEBUG)
    else()
        target_compile_options(diagnostic-probe PRIVATE -UNDEBUG)
    endif()
    add_test(NAME diagnostics COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/diagnostic-test.py $<TARGET_FILE:diagnostic-probe>)
    add_executable(model-probe tests/model-probe.c)
    target_link_libraries(model-probe PRIVATE chip-sequencer)
    add_test(NAME controls COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/control-test.py $<TARGET_FILE:model-probe>)
    add_test(NAME sequence COMMAND sequence-test ${CMAKE_CURRENT_SOURCE_DIR}/examples/sequence.json)
    add_test(NAME compile COMMAND compile-test ${CMAKE_CURRENT_SOURCE_DIR}/examples/performance.json ${CMAKE_CURRENT_BINARY_DIR}/compiled.json)
endif()
option(CHIP_SEQUENCER_BUILD_EXAMPLES "Build standalone PCM renderer" OFF)
if(CHIP_SEQUENCER_BUILD_EXAMPLES)
    add_executable(chip-render examples/render.c)
    target_link_libraries(chip-render PRIVATE chip-sequencer)
endif()
